156    @field_validator('evidence_domains')
157    def validate_domains(cls, v):
158        """Check domain are valid, catch LLM hallucinations."""
159        if not v:
160            return v
161           
162        domain_pattern = re.compile(r'^(?=.{1,253}$)(?!-)[A-Za-z0-9-]{1,63}(?<!-)(\.[A-Za-z]{2,})+$')
163        
164        validated = []
165        for domain in v:
166            clean = domain.replace('http://', '').replace('https://', '').split('/')[0]
167            
168            if domain_pattern.match(clean) or re.match(r'^\d{1,3}(\.\d{1,3}){3}$', clean):  # IP address
169                validated.append(clean)
170            else:
171                logger.warning(f"Invalid domain format: {domain}")
172        
173        return validated

178        if not v:

268    def _adjust_severity_with_confidence(self, base_severity: str) -> str:
269        """Adjust severity based on model confidence and evidence quality."""
270        # Confidence thresholds
271        if self.confidence < 0.3:
272            severity_map = {"critical": "high", "high": "medium", "medium": "low", "low": "info"}       # Low confidence - downgrade severity
273            return severity_map.get(base_severity, "info")
274        
275        elif self.confidence < 0.6:
276            if base_severity == "critical":                                                             # Medium - keep or slightly downgrade
277                return "high"
278            return base_severity
279        
280        elif self.confidence >= 0.8:
281            if base_severity == "high" and self._has_strong_evidence():                                 # High - consider upgrading
282                return "critical"
283            elif base_severity == "medium" and self._has_strong_evidence():
284                return "high"
285        
286        return base_severity

288    def _adjust_severity_with_context(self, current_severity: str) -> str:
289        """Adjust severity based on contextual factors."""
290        if self.context_legitimacy_score and self.context_legitimacy_score > 0.7:                      # Override Context legitimacy if behavior is legitimate in context
291            severity_map = {"critical": "low", "high": "info", "medium": "info", "low": "info"}
292            return severity_map.get(current_severity, "info")
293        
294        if self.extraction_stage == "structure" and current_severity != "critical":
295            return "high" 
296        
297        if self.category in [BehaviorCategory.ANTI_DEBUGGING, BehaviorCategory.ANTI_ANALYSIS]:         # Anti-analysis behaviors tend to be more severe
298            if current_severity == "medium":
299                return "high"
300        
301        return current_severity

318        return evidence_score >= 2

479class PatternDetectionMetrics(BaseModel):
480    """Performance metrics on validation set (training phase)."""
481    true_positives: int = 0
482    false_positives: int = 0
483    true_negatives: int = 0              
484    false_negatives: int = 0             
485    precision: Optional[float] = None    # TP / (TP + FP)
486    recall: Optional[float] = None       # TP / (TP + FN)
487    f1_score: Optional[float] = None     # Harmonic mean of precision & recall
488    
489    @model_validator(mode='after')
490    def compute_metrics(self) -> 'PatternDetectionMetrics':
491        """Auto-compute metrics from confusion matrix."""
492        if self.true_positives + self.false_positives > 0:
493            self.precision = self.true_positives / (self.true_positives + self.false_positives)
494        if self.true_positives + self.false_negatives > 0:
495            self.recall = self.true_positives / (self.true_positives + self.false_negatives)
496        if self.precision and self.recall and self.precision + self.recall > 0:
497            self.f1_score = 2 * (self.precision * self.recall) / (self.precision + self.recall)
498        return self
