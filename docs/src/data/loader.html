 27def _add_mode(directory: str):
 28    """
 29    Fix directory and file permissions recursively.
 30    
 31    Args:
 32        directory: Folder path to fix permissions for
 33    """
 34    if not os.path.exists(directory):
 35        return
 36    
 37    try:
 38        for dirpath, dirnames, filenames in os.walk(directory):
 39            # Fix directory permissions (rwxr-xr-x = 755)
 40            for dirname in dirnames:
 41                dir_path = os.path.join(dirpath, dirname)
 42                if not os.access(dir_path, os.R_OK | os.W_OK | os.X_OK):
 43                    try:
 44                        os.chmod(dir_path, 0o755)
 45                    except Exception as e:
 46                        logger.warning(f"Could not fix permissions for directory {dir_path}: {e}")
 47              
 48            # Fix file permissions (rw-r--r-- = 644)          
 49            for filename in filenames:
 50                file_path = os.path.join(dirpath, filename)
 51                if not os.access(file_path, os.R_OK | os.W_OK):
 52                    try:
 53                        os.chmod(file_path, 0o644)
 54                    except Exception as e:
 55                        logger.warning(f"Could not fix permissions for file {file_path}: {e}")
 56
 57    except Exception as e:
 58        logger.error(f"Error while fixing permissions in {directory}: {e}")

 60def _decompress_tarballs(
 61    source_dir: Path,
 62    dest_dir: Path,
 63    use_cache: bool = False,
 64    show_progress: bool = True
 65) -> Tuple[Path, bool]:
 66    """
 67    Decompress .tar.gz and .tgz files to a destination directory.
 68        
 69    Args:
 70        source_dir: Directory containing tarballs
 71        dest_dir: Where to extract packages
 72        use_cache: If True and dest already exists, skip decompression
 73        show_progress: Show progress bar
 74    
 75    Returns:
 76        Tuple of (extraction_path, was_extracted)
 77        - extraction_path: Path to directory with extracted packages
 78        - was_extracted: True if extraction was performed, False if cached
 79    """
 80    marker_file = dest_dir / EXTRACTION_MARKER
 81    
 82    if use_cache and marker_file.exists():
 83        logger.info(f"Using cached decompressed dataset: {dest_dir}")
 84        return dest_dir, False
 85    
 86    if not dest_dir.exists():
 87        dest_dir.mkdir(parents=True, exist_ok=True)
 88    
 89    
 90    tarball_files = [
 91        f for f in source_dir.iterdir()
 92        if f.is_file() and (f.suffix in ['.gz', '.tgz'] or f.name.endswith('.tar.gz'))
 93    ]
 94    
 95    if not tarball_files:
 96        logger.warning(f"No tarballs found in {source_dir}")
 97        return dest_dir, False
 98    
 99    logger.info(f"Found {len(tarball_files)} tarballs to decompress")
100    
101    if dest_dir.exists() and not marker_file.exists():
102        logger.info("Incomplete extraction found, cleaning up...")
103        shutil.rmtree(dest_dir)
104        dest_dir.mkdir(parents=True)
105    
106    iterable = tqdm(tarball_files, desc="Decompressing tarballs") if show_progress else tarball_files
107    
108    failed = []
109    successful = 0
110    
111    for tarball_path in iterable:
112        # Get clean package name from tarball (remove .tar/.gz extension)
113        package_name = tarball_path.stem
114        if package_name.endswith('.tar'):
115            package_name = package_name[:-4]
116            
117        package_name = "".join(c for c in package_name if c.isalnum() or c in ('-', '_','.','@'))
118            
119        temp_path = dest_dir / package_name
120        
121        try:
122            temp_path.mkdir(parents=True, exist_ok=True)
123            with tarfile.open(tarball_path, 'r:gz') as tar:
124                def is_safe_path(member):
125                    if member.isdir():
126                        return True
127                    member_path = Path(temp_path) / member.name
128                    try:
129                        member_path.resolve().relative_to(temp_path.resolve())      # Ensure member path is inside temp_package_path
130                        return True
131                    except ValueError:
132                        logger.warning(f"Unsafe path detected in tarball {tarball_path}: {member.name}")
133                        return False
134                
135                safe_members = [m for m in tar.getmembers() if is_safe_path(m)]
136                if safe_members:
137                    tar.extractall(path=temp_path, members=safe_members)
138                else:
139                    raise ValueError(f"No safe paths in {tarball_path.name} to extract")
140                
141                successful += 1
142
143        except Exception as e:
144            logger.error(f"Failed to decompress for {package_name}: {e}")
145            failed.append(tarball_path.name)
146            
147            if temp_path.exists():
148                shutil.rmtree(temp_path, ignore_errors=True)
149                
150    _add_mode(str(dest_dir))
151    
152    if successful > 0:
153        marker_file.write_text(
154            f"Extraction completed at: {datetime.now()}\n"
155            f"Source: {source_dir}\n"
156            f"Success: {successful}/{len(tarball_files)}\n"
157            f"Failed: {len(failed)}\n"
158            f"Cache version: {CACHE_VERSION}"
159        )
160        logger.info(f"Extraction complete: {successful}/{len(tarball_files)} successful")
161
162        if failed:
163            logger.warning("Failed packages:\n" + "\n".join(failed))
164
165        
166        return dest_dir, True
167    else:
168        logger.error("No packages were successfully extracted!")
169        return dest_dir, False

412        """

480        """

482        
483        """
484        
485        scripts = package_data.get("scripts", {})

412    def validate_dataset(self, sample_size: int = 10) -> Dict[str, Any]:
413        """
414        Validate dataset structure and extraction.
415        """
416        logger.info(f"Validating dataset with {sample_size} samples...")
417        
418        try:
419            extracted_path = self.ensure_extracted("malicious", use_cache=True, show_progress=False)
420        except Exception as e:

422                "valid": False,
423                "error": f"Extraction failed: {e}"
424            }
425        
426        all_dirs = [d for d in extracted_path.iterdir() if d.is_dir()]
427        package_dirs = sorted(all_dirs)[:sample_size]
428
429        if not package_dirs:
430            return {
431                "valid": False,
432                "error": "No extracted packages found"
433            }
434

436        valid_count = 0

438        for pkg_dir in package_dirs:
439            package_subdir = pkg_dir / "package"

444                
445            if not package_json_path.exists():
446                # Look for package.json in subdirectories (common in extracted tarballs)
447                package_json_candidates = list(pkg_dir.rglob("package.json"))
448                if not package_json_candidates:
449                    issues.append(f"{pkg_dir.name}: No package.json found")
450                    continue
451                package_json_path = package_json_candidates[0]
452            
453            try:
454                with open(package_json_path) as f:

456            except (json.JSONDecodeError, Exception) as e:
457                issues.append(f"{pkg_dir.name}: Invalid package.json - {e}")
458                continue
459                
460            if not package_data.get("name"):
461                issues.append(f"{pkg_dir.name}: Missing package name")
462                continue
463                
464            valid_count += 1
465        
466        is_valid = valid_count >= max(1, len(package_dirs) * 0.8)       # At least 80% valid
467        status = "VALID" if is_valid else "INVALID"
468        
469        logger.info(f"{status}: {valid_count}/{len(package_dirs)} packages valid")
470        
471        return {

473            "total_checked": len(package_dirs),

480    def _extract_install_script(self, package_data: dict) -> Optional[str]:
481        """
482        Extract install script from package.json data.
483        
484        """
485        
486        scripts = package_data.get("scripts", {})
487        install_hooks = {}
488        
489        for hook in ["install", "preinstall", "postinstall", "prepublish"]:
490            if hook in scripts:
491                install_hooks[hook] = scripts[hook]
492                
493        if install_hooks:
494            return "\n".join(f"{k}: {v}" for k, v in install_hooks.items())
495        
496        return None
